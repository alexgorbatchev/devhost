Stack configuration

The whole stack.

Source configuration, including stack settings and all service tables.

OverviewConfigurationDependencies

devhost.toml

TOML / source 
name = "devhost"

[devtools.shortcuts]
restartServices = "alt+ctrl+r"

# The browser `just dev-browser` starts; DEVHOST_BROWSER_ENDPOINT is empty, and native React access off, until it is set.
[devtools.browser]
endpoint = "{{ env.DEVHOST_BROWSER_ENDPOINT }}"

[services.storybook]
primary = true
command = ["sh", "-c", "exec ./node_modules/.bin/storybook dev --ci --port \"$PORT\""]
cwd = "packages/devhost-ui"
port = "auto"
host = "devhost-storybook{{ env.DEVHOST_DEV_EXT }}"

[services.storybook.env]
__VITE_ADDITIONAL_SERVER_ALLOWED_HOSTS = "{{ services.storybook.host }}"

[services.playground-backend]
command = ["just", "run"]
cwd = "packages/playground/backend"
port = "auto"
host = "devhost-playground{{ env.DEVHOST_DEV_EXT }}"
path = "/api/*"

[services.playground-frontend]
command = ["just", "run"]
cwd = "packages/playground/frontend"
bindHost = "127.0.0.1"
port = "auto"
host = "playground{{ env.DEVHOST_DEV_EXT }}"
path = "/"
# Bun HTML/HMR requires Host and Origin to use bindHost plus the assigned port.
proxyLocalOrigin = true
watch = ["src/"]

[services.playground-frontend.env]
NODE_ENV = "development"

# Keep architecture documentation available with every service selection.
[services.archify]
alwaysStart = true
command = ["just", "archify", "run"]
port = "auto"
bindHost = "127.0.0.1"
host = "devhost-archify{{ env.DEVHOST_DEV_EXT }}"

[services.archify.health]
http = "/"
